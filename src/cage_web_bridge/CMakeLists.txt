project(cage_web_bridge)
find_package(catkin REQUIRED COMPONENTS cage_geo cage_msgs mavros_msgs rospy sensor_msgs std_msgs)
catkin_package(CATKIN_DEPENDS cage_geo cage_msgs mavros_msgs rospy sensor_msgs std_msgs)

install(DIRECTORY www config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test)
endif()
